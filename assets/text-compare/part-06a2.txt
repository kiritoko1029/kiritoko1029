  function lineText(text, starts, index) {
    if (index == null || index < 0 || index >= starts.length) return null;
    const start = starts[index];
    let end = index + 1 < starts.length ? starts[index + 1] : text.length;
    if (end > start && text.charCodeAt(end - 1) === 10) end--;
    if (end > start && text.charCodeAt(end - 1) === 13) end--;
    return text.slice(start, end);
  }
  function findBlockByRow(rowIndex) {
    let low = 0, high = state.blocks.length - 1;
    while (low <= high) {
      const middle = (low + high) >>> 1, block = state.blocks[middle];
      if (rowIndex < block.rowStart) high = middle - 1;
      else if (rowIndex >= block.rowStart + block.length) low = middle + 1;
      else return block;
    }
    return null;
  }
  function findDiffBlock(position) {
    const indexes = state.diffBlockIndexes;
    let low = 0, high = indexes.length - 1;
    while (low <= high) {
      const middle = (low + high) >>> 1, block = state.blocks[indexes[middle]];
      if (position < block.diffStart) high = middle - 1;
      else if (position >= block.diffStart + block.length) low = middle + 1;
      else return block;
    }
    return null;
  }
  function visiblePositionToRow(position) {
    if (!$('onlyDiff').checked) return position;
    const block = findDiffBlock(position);
    return block ? block.rowStart + position - block.diffStart : -1;
  }
