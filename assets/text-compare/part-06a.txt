  function toast(message, error = false, duration) {
    clearTimeout(toastTimer);
    $('toast').textContent = message;
    $('toast').classList.toggle('error', error);
    $('toast').hidden = false;
    toastTimer = setTimeout(() => { $('toast').hidden = true; }, duration || (error ? 6500 : 3600));
  }
  function setProgress(stage, progress) {
    if (!busy) return;
    const percentage = Math.max(1, Math.min(99, Math.round((progress || 0) * 100)));
    $('compareLabel').textContent = `${stage || '正在对比'} ${percentage}%`;
    $('liveStatus').textContent = `${stage || '正在对比'}，${percentage}%`;
  }
  function setBusy(value) {
    busy = value;
    $('compareBtn').setAttribute('aria-busy', String(value));
    $('compareBtn').classList.toggle('is-cancel', value);
    $('compareBtn').title = value ? '点击取消当前对比' : '';
    $('compareLabel').textContent = value ? '正在准备…' : '开始对比';
    $('compareIcon').classList.toggle('spinning', value);
    $('compareIcon').querySelector('use').setAttribute('href', value ? '#i-loader' : '#i-compare');
  }
  function disposeWorker() {
    if (worker) { worker.terminate(); worker = null; }
    if (workerUrl) { URL.revokeObjectURL(workerUrl); workerUrl = null; }
  }
  function cancelPending(showMessage = false) {
    const wasBusy = busy;
    taskId++;
    disposeWorker();
    setBusy(false);
    if (showMessage && wasBusy) {
      toast('已取消本次对比，文本仍然保留。');
      $('liveStatus').textContent = '已取消本次对比。';
    }
  }
  function invalidate(message = '文本已更新，重新对比以查看差异。') {
    cancelPending(false);
    if (state) {
      state = null; $('diffRows').replaceChildren();
      $('results').hidden = true; $('emptyState').hidden = false;
      $('emptyTitle').textContent = '准备好新的对比';
      $('emptyDescription').textContent = message;
      $('liveStatus').textContent = message;
    }
    scheduleCounts();
  }
  function options() {
    return { mode: document.querySelector('input[name="mode"]:checked').value, ignoreSpace: $('ignoreSpace').checked, ignoreCase: $('ignoreCase').checked };
  }
  function optionDescription(opts) {
    const values = [opts.mode === 'line' ? '整行对比' : '字符级对比'];
    if (opts.ignoreSpace) values.push('忽略行内空白');
    if (opts.ignoreCase) values.push('忽略大小写');
    values.push('统一换行格式');
    return values.join(' · ');
  }

  function lineText(text, starts, index) {
    if (index == null || index < 0 || index >= starts.length) return null;
    const start = starts[index];
    let end = index + 1 < starts.length ? starts[index + 1] : text.length;
    if (end > start && text.charCodeAt(end - 1) === 10) end--;
    if (end > start && text.charCodeAt(end - 1) === 13) end--;
    return text.slice(start, end);
  }
  function findBlockByRow(rowIndex) {
    let low = 0, high = state.blocks.length - 1;
    while (low <= high) {
      const middle = (low + high) >>> 1, block = state.blocks[middle];
      if (rowIndex < block.rowStart) high = middle - 1;
      else if (rowIndex >= block.rowStart + block.length) low = middle + 1;
      else return block;
    }
    return null;
  }
  function findDiffBlock(position) {
    const indexes = state.diffBlockIndexes;
    let low = 0, high = indexes.length - 1;
    while (low <= high) {
      const middle = (low + high) >>> 1, block = state.blocks[indexes[middle]];
      if (position < block.diffStart) high = middle - 1;
      else if (position >= block.diffStart + block.length) low = middle + 1;
      else return block;
    }
    return null;
  }
  function visiblePositionToRow(position) {
    if (!$('onlyDiff').checked) return position;
    const block = findDiffBlock(position);
    return block ? block.rowStart + position - block.diffStart : -1;
  }
