unction splitLines(text) {
    if (text === '') return { lines: [], starts: new Uint32Array(0) };
    const lines = [], starts = [];
    let start = 0;
    for (let i = 0; i < text.length; i++) {
      const code = text.charCodeAt(i);
      if (code !== 10 && code !== 13) continue;
      starts.push(start);
      lines.push(text.slice(start, i));
      if (code === 13 && text.charCodeAt(i + 1) === 10) i++;
      start = i + 1;
    }
    starts.push(start);
    lines.push(text.slice(start));
    return { lines, starts: Uint32Array.from(starts) };
  }

  function appendSegment(output, type, aStart, bStart, length) {
    if (length <= 0) return;
    const last = output[output.length - 1];
    const contiguous = last && last.type === type && (
      type === 'equal'
        ? last.aStart + last.length === aStart && last.bStart + last.length === bStart
        : type === 'remove'
          ? last.aStart + last.length === aStart && last.bStart === bStart
          : last.bStart + last.length === bStart && last.aStart === aStart
    );
    if (contiguous) last.length += length;
    else output.push({ type, aStart, bStart, length });
  }

  function appendCoarse(output, aStart, aEnd, bStart, bEnd, budget) {
    budget.coarse = true;
    appendSegment(output, 'remove', aStart, bStart, aEnd - aStart);
    appendSegment(output, 'add', aEnd, bStart, bEnd - bStart);
  }

  function backtrackSegments(trace, n, m, offset, aOffset, bOffset) {
    const reversed = [];
    let x = n, y = m;
    for (let d = trace.length - 1; d >= 0; d--) {
      const v = trace[d], k = x - y;
      const previousK = k === -d || (k !== d && v[offset + k - 1] < v[offset + k + 1]) ? k + 1 : k - 1;
      const previousX = v[offset + previousK];
      const previousY = previousX - previousK;
      const snake = Math.min(x - previousX, y - previousY);
      if (snake > 0) {
        reversed.push({ type: 'equal', aStart: aOffset + x - snake, bStart: bOffset + y - snake, length: snake });
        x -= snake; y -= snake;
      }
      if (d === 0) break;
      if (x === previousX) {
        y--;
        reversed.push({ type: 'add', aStart: aOffset + x, bStart: bOffset + y, length: 1 });
      } else {
        x--;
        reversed.push({ type: 'remove', aStart: aOffset + x, bStart: bOffset + y, length: 1 });
      }
    }
    const output = [];
    for (let i = reversed.length - 1; i >= 0; i--) {
      const segment = reversed[i];
      appendSegment(output, segment.type, segment.aStart, segment.bStart, segment.length);
    }
    return output;
  }

  function myersRange(a, b, aStart, aEnd, bStart, bEnd, budget, maxDistance) {
    const n = aEnd - aStart, m = bEnd - bStart;
    if (!n) return [{ type: 'add', aStart, bStart, length: m }];
    if (!m) return [{ type: 'remove', aStart, bStart, length: n }];
    const limit = Math.min(n + m, maxDistance);
    if (Math.abs(n - m) > limit || budget.remaining <= 0) return null;
    const offset = limit + 1;
    const v = new Int32Array(2 * limit + 3);
    v.fill(-1);
    v[offset + 1] = 0;
    const trace = [];
    for (let d = 0; d <= limit; d++) {
      trace.push(v.slice());
      for (let k = -d; k <= d; k += 2) {
        if (--budget.remaining < 0) return null;
        const index = offset + k;
        let x = k === -d || (k !== d && v[index - 1] < v[index + 1]) ? v[index + 1] : v[index - 1] + 1;
        let y = x - k;
        while (x < n && y < m && a[aStart + x] === b[bStart + y]) {
          x++; y++;
          if (--budget.remaining < 0) return null;
        }
        v[index] = x;
        if (x >= n && y >= m) return backtrackSegments(trace, n, m, offset, aStart, bStart);
      }
    }
    return null;
  }
