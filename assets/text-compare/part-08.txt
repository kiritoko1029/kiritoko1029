Error('这个文件看起来不是普通文本文件。');
    try {
      return { text: new TextDecoder('utf-8', { fatal: true }).decode(bytes), encoding };
    } catch (error) {
      try { return { text: new TextDecoder('gb18030').decode(bytes), encoding: 'GB18030 / GBK' }; }
      catch (fallbackError) { return { text: new TextDecoder('utf-8').decode(bytes), encoding: 'UTF-8（含替换字符）' }; }
    }
  }
  function readLocalFile(file, side) {
    if (!file) return;
    const input = side === 'left' ? leftInput : rightInput;
    const sourceLabel = $(side === 'left' ? 'leftSource' : 'rightSource');
    const reader = new FileReader();
    importing = true;
    sourceLabel.textContent = `正在读取 ${file.name}…`;
    sourceLabel.title = file.name;
    reader.onprogress = event => {
      if (event.lengthComputable) sourceLabel.textContent = `正在读取 ${file.name} · ${Math.round(event.loaded / event.total * 100)}%`;
    };
    reader.onerror = () => {
      importing = false; updateSourceLabel(side);
      toast(`无法读取“${file.name}”。`, true);
    };
    reader.onload = () => {
      try {
        const decoded = decodeTextBuffer(reader.result);
        input.value = decoded.text;
        sourceInfo[side] = { name: file.name, size: file.size, path: `${file.name} · ${decoded.encoding}` };
        updateSourceLabel(side);
        invalidate(`已导入 ${file.name}，点击「开始对比」计算差异。`);
        scheduleCounts(true);
        toast(`已导入 ${file.name}（${fileSize(file.size)}，${decoded.encoding}）。`);
      } catch (error) {
        sourceInfo[side] = null; updateSourceLabel(side);
        toast(error.message || `无法读取“${file.name}”。`, true);
      } finally { importing = false; }
    };
    reader.readAsArrayBuffer(file);
  }
  function setupFileImport(side) {
    const button = $(side === 'left' ? 'importLeft' : 'importRight');
    const picker = $(side === 'left' ? 'leftFile' : 'rightFile');
    const editor = $(side === 'left' ? 'leftEditor' : 'rightEditor');
    button.addEventListener('click', () => { picker.value = ''; picker.click(); });
    picker.addEventListener('change', () => readLocalFile(picker.files && picker.files[0], side));
    editor.addEventListener('dragover', event => {
      if (!event.dataTransfer || !Array.from(event.dataTransfer.types || []).includes('Files')) return;
      event.preventDefault(); editor.classList.add('is-dragover');
    });
    editor.addEventListener('dragleave', event => { if (!editor.contains(event.relatedTarget)) editor.classList.remove('is-dragover'); });
    editor.addEventListener('drop', event => {
      editor.classList.remove('is-dragover');
      const file = event.dataTransfer && event.dataTransfer.files && event.dataTransfer.files[0];
      if (!file) return;
      event.preventDefault(); readLocalFile(file, side);
    });
  }
